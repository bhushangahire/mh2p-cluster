# Per-car settings baked into the installed cluster_config.json by install.sh.
#
# Copy this file to  Mods/ClusterIntegration/config_overrides.txt  (same
# folder) and uncomment what you need. One  key=value  per line; the value is
# written exactly as JSON (true / false / a number / "a string"). Keys are
# looked up in the top-level "config" block of cluster_config.json; the
# inline mirror and gal_h264 objects of that block are reached as
# gal_h264.mode, mirror.zoomX and so on. carConfig entries are not touched.
# Re-running the installer without config_overrides.txt reinstalls the
# shipped config.
#
# Right-hand-drive car (UK, Ireland, Japan, Australia, ...): forces the
# roundabout angle correction and Waze keep-lane side without relying on the
# car's driver-side property or the country table.
#forceRHD=true
#
# Show yards below 0.1 mi instead of feet when the car is set to miles
# (only matters while imperialSmallUnit is "auto" and the country lookup fails).
#imperialSmallUnit="yards"
#
# Always use miles/yards regardless of the car's unit setting.
#forceImperial=true
#
# Turn-by-turn only (no cluster map video), the same as diagnostic mode does.
#enableMapRender=false
#
# Text only instead of text + bargraph if the virtual cockpit draws both badly.
#bargraphMode="distance"

# Video window of the Android Auto map on the cluster (the global gal_h264
# object, used when carConfig has no entry for your car). BLOCK.KEY form.
#   mode: letter (fit, black bars), fill (cover, crops top/bottom), stretch
#   codecRes: 480, 720 or 1080 (the stream size asked from the phone)
#   zoomX/zoomY > 1 enlarge (crop), panX/panY shift the visible part
# e-tron (1920x722 cockpit): codecRes=1080 with mode="fill" shows the centre
# 722 rows (179-900) of the 1080-row frame pixel for pixel across the full
# width, and the cluster entry built into gal_cluster.so (modkit/template/
# template.txt) tells the phone to lay its UI out inside exactly that band
# and clear of the Audi info bar, so keep panY=0 with it. (The earlier
# panY=0.248 recipe only panned to where Google had put the car marker.)
# letter (the default) pillarboxes 16:9 frames; see the README.
#gal_h264.mode="fill"
#gal_h264.codecRes=1080
#gal_h264.panY=0
#gal_h264.zoomX=1.0
#gal_h264.zoomY=1.0
#gal_h264.panX=0.0
# Same keys for the mirror (screen-capture) path: mirror.mode, mirror.zoomX ...
